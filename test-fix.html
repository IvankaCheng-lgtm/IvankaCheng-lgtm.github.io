🔧 修復細節

新增的 hideAllScreens() 函數

function hideAllScreens() {
    const allScreens = [
        welcomeScreen,
        profileSetupScreen,
        mainScreen,
        cameraScreen,
        resultScreen,
        document.getElementById('nutrition-screen'),
        document.getElementById('profile-screen'),
        document.getElementById('ai-advice-screen'),
        document.getElementById('history-screen'),
        document.getElementById('food-library-screen')
    ];
    
    allScreens.forEach(screen => {
        if (screen) {
            screen.classList.add('hidden');
            screen.style.display = 'none';
        }
    });
}

頁面切換模式

// 新的頁面切換模式
function switchToScreen(targetScreen) {
    // 1. 先隱藏所有頁面
    hideAllScreens();
    
    // 2. 顯示目標頁面
    if (targetScreen) {
        targetScreen.classList.remove('hidden');
        targetScreen.style.display = 'block'; // 或 'flex' 等適當的顯示方式
    }
}
